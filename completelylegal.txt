<script>
  const STORE_KEY = '1a3_notes_files';
  let staged = [];

  function getFiles() {
    try { return JSON.parse(localStorage.getItem(STORE_KEY)) || []; } catch { return []; }
  }
  function saveFiles(arr) { localStorage.setItem(STORE_KEY, JSON.stringify(arr)); }

  function ext(name) {
    const parts = name.split('.');
    return parts.length > 1 ? parts.pop().slice(0,4) : 'file';
  }

  function fmt(bytes) {
    if (bytes < 1024) return bytes + ' B';
    if (bytes < 1048576) return (bytes/1024).toFixed(1) + ' KB';
    return (bytes/1048576).toFixed(1) + ' MB';
  }

  function toast(msg) {
    const el = document.getElementById('toast');
    el.textContent = msg;
    el.classList.add('show');
    setTimeout(() => el.classList.remove('show'), 2800);
  }

  function renderFiles() {
    const files = getFiles();
    const el = document.getElementById('fileList');
    if (!files.length) {
      el.innerHTML = '<div class="empty-state"><p class="empty-label">No files uploaded yet — be the first to share</p></div>';
      return;
    }
    el.innerHTML = '<div class="file-list">' + files.map((f, i) => `
      <div class="file-item">
        <div class="file-ext">${ext(f.name)}</div>
        <div class="file-info">
          <div class="file-name">${f.name}</div>
          <div class="file-meta">${f.subject ? f.subject + ' · ' : ''}${f.uploader} · ${f.date} · ${f.size}</div>
        </div>
        <div class="file-actions">
          <button class="btn-action" onclick="downloadFile(${i})">Download</button>
          <button class="btn-action btn-del" onclick="deleteFile(${i})">Remove</button>
        </div>
      </div>`).join('') + '</div>';
  }

  function commitUpload() {
    if (!staged.length) { toast('Please select at least one file first.'); return; }
    const uploader = document.getElementById('uploaderName').value.trim() || 'Anonymous';
    const subject  = document.getElementById('fileSubject').value.trim();
    const files    = getFiles();
    const now      = new Date().toLocaleDateString('en-SG', {day:'numeric',month:'short',year:'numeric'});
    staged.forEach(f => files.unshift({ name: f.name, size: fmt(f.size), uploader, subject, date: now, data: f.data, type: f.type }));
    saveFiles(files);
    staged = [];
    document.getElementById('fileInput').value = '';
    toast('Files added successfully.');
    renderFiles();
  }

  function deleteFile(i) {
    const files = getFiles();
    files.splice(i, 1);
    saveFiles(files);
    renderFiles();
    toast('File removed.');
  }

  function downloadFile(i) {
    const f = getFiles()[i];
    if (!f.data) { toast('No download data — file was added before download support.'); return; }
    const a = document.createElement('a');
    a.href = f.data;
    a.download = f.name;
    a.click();
  }

  // File input change
  document.getElementById('fileInput').addEventListener('change', function() {
    handleFiles(this.files);
  });

  function handleFiles(fileList) {
    staged = [];
    Array.from(fileList).forEach(file => {
      const reader = new FileReader();
      reader.onload = e => staged.push({ name: file.name, size: file.size, type: file.type, data: e.target.result });
      reader.readAsDataURL(file);
    });
    toast(fileList.length + ' file(s) selected. Click "Add Files" to confirm.');
  }

  // Drag and drop
  const dz = document.getElementById('dropZone');
  dz.addEventListener('dragover', e => { e.preventDefault(); dz.classList.add('dragover'); });
  dz.addEventListener('dragleave', () => dz.classList.remove('dragover'));
  dz.addEventListener('drop', e => {
    e.preventDefault();
    dz.classList.remove('dragover');
    handleFiles(e.dataTransfer.files);
  });

  renderFiles();

  function dueClass(dueStr) {
    if (!dueStr) return '';
    const due  = new Date(dueStr);
    const now  = new Date();
    const diff = (due - now) / (1000 * 60 * 60 * 24);
    if (diff < 0)  return 'due-soon';
    if (diff < 1)  return 'due-today';
    if (diff < 3)  return 'due-soon';
    return '';
  }

  function dueLabel(dueStr) {
    if (!dueStr) return '';
    const due = new Date(dueStr);
    const now = new Date();
    const diff = Math.ceil((due - now) / (1000*60*60*24));
    if (diff < 0)  return 'Overdue';
    if (diff === 0) return 'Due Today';
    if (diff === 1) return 'Due Tomorrow';
    return 'Due ' + due.toLocaleDateString('en-SG',{day:'numeric',month:'short'});
  }

<script>
  const STORE_KEY = '1a3_events';
  let activeFilter = 'All';

  function getEvents() { try { return JSON.parse(localStorage.getItem(STORE_KEY)) || []; } catch { return []; } }
  function saveEvents(arr) { localStorage.setItem(STORE_KEY, JSON.stringify(arr)); }

  function toast(msg) {
    const el = document.getElementById('toast');
    el.textContent = msg;
    el.classList.add('show');
    setTimeout(() => el.classList.remove('show'), 2800);
  }

  function setFilter(f, btn) {
    activeFilter = f;
    document.querySelectorAll('.filter-btn').forEach(b => b.classList.remove('active'));
    btn.classList.add('active');
    renderEvents();
  }

  function countdown(dateStr) {
    if (!dateStr) return { label: '', cls: 'upcoming' };
    const d    = new Date(dateStr);
    const now  = new Date();
    now.setHours(0,0,0,0);
    d.setHours(0,0,0,0);
    const diff = Math.round((d - now) / (1000*60*60*24));
    if (diff < 0)  return { label: Math.abs(diff) + ' days ago', cls: 'past' };
    if (diff === 0) return { label: 'Today', cls: 'today' };
    if (diff === 1) return { label: 'Tomorrow', cls: 'soon' };
    if (diff <= 7)  return { label: 'In ' + diff + ' days', cls: 'soon' };
    return { label: 'In ' + diff + ' days', cls: 'upcoming' };
  }

  function isPast(dateStr) {
    if (!dateStr) return false;
    const d = new Date(dateStr); d.setHours(0,0,0,0);
    const n = new Date(); n.setHours(0,0,0,0);
    return d < n;
  }

  function priorityLabel(p) {
    return { critical: 'Critical', high: 'Important', normal: 'Normal', info: 'Info' }[p] || p;
  }

  function renderEvents() {
    const all = getEvents();
    let list = all;
    if (activeFilter === 'upcoming') list = all.filter(e => !isPast(e.date));
    else if (activeFilter === 'past') list = all.filter(e => isPast(e.date));
    else if (activeFilter !== 'All') list = all.filter(e => e.priority === activeFilter);

    // Sort: upcoming first by date, then past
    list = list.slice().sort((a,b) => {
      const da = a.date ? new Date(a.date) : new Date('9999-01-01');
      const db = b.date ? new Date(b.date) : new Date('9999-01-01');
      return da - db;
    });

    const el = document.getElementById('eventsGrid');
    if (!list.length) {
      el.innerHTML = '<div class="empty-state"><p class="empty-label">No events yet — pin one above</p></div>';
      return;
    }

    el.innerHTML = '<div class="events-grid">' + list.map(e => {
      const idx = all.indexOf(e);
      const cd  = countdown(e.date);
      const d   = e.date ? new Date(e.date) : null;
      const dayNum   = d ? d.getDate() : '';
      const monthStr = d ? d.toLocaleDateString('en-SG',{month:'short',year:'numeric'}) : '';
      return `<div class="event-card priority-${e.priority || 'normal'}">
        <div class="event-top">
          <div class="event-type">${e.category || priorityLabel(e.priority)}</div>
          ${d ? `<div class="event-date-display"><div class="event-day">${dayNum}</div><div class="event-month">${monthStr}</div></div>` : ''}
        </div>
        <div class="event-title">${e.title}</div>
        ${e.body ? `<div class="event-body">${e.body}</div>` : ''}
        <div class="event-footer">
          <div class="event-author">${e.author ? 'By ' + e.author : ''}</div>
          <div style="display:flex;align-items:center;gap:12px;">
            ${e.date ? `<span class="countdown ${cd.cls}">${cd.label}</span>` : ''}
            <button class="btn-del" onclick="deleteEvent(${idx})">Remove</button>
          </div>
        </div>
      </div>`;
    }).join('') + '</div>';
  }

  function addEvent() {
    const title    = document.getElementById('evTitle').value.trim();
    const date     = document.getElementById('evDate').value;
    const priority = document.getElementById('evPriority').value;
    const category = document.getElementById('evCategory').value;
    const author   = document.getElementById('evAuthor').value.trim();
    const body     = document.getElementById('evBody').value.trim();
    if (!title) { toast('Please enter an event title.'); return; }
    const events = getEvents();
    events.unshift({ title, date, priority, category, author, body });
    saveEvents(events);
    document.getElementById('evTitle').value    = '';
    document.getElementById('evDate').value     = '';
    document.getElementById('evBody').value     = '';
    document.getElementById('evCategory').value = '';
    toast('Event pinned to the board.');
    renderEvents();
  }

  function deleteEvent(i) {
    const events = getEvents();
    events.splice(i, 1);
    saveEvents(events);
    renderEvents();
    toast('Event removed.');
  }

  renderEvents();
</script>
